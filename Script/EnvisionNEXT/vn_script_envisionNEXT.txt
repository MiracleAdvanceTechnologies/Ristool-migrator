UPDATE "PMS_RPT_REGION"  
SET  "DISP_NAME"=local_disp_name, 
     "SINGULAR"=CASE  WHEN "SINGULAR" like '%$%' THEN '$'
      ELSE local_disp_name

AND  "PMS_RPT_REGION"."RPT_REGION_ID" NOT IN
(
select "RPT_REGION_ID"  from "PMS_RPT_REGION" prr where "NAME" like '%cm%' or "NAME" ='Normal' or "NAME" ='Abnormal' or "NAME"='N/A'

"PMS_RPT_DIAGONOSIS"."RPT_DIAGONOSIS_ID" NOT IN
(
select "RPT_DIAGONOSIS_ID" PT from "PMS_RPT_DIAGONOSIS" prd  where ("CLASS3" =0 and "CLASS1" !=0 and "CLASS2" !=0) or ("CLASS3" !=0 and "CLASS1" !=0 and "CLASS2" =0)
